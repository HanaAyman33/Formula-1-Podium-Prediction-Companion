//1. Query 1
MATCH (res:Result)-[:FOR_RACE]->(r:Race)-[:IN_SEASON]->(:Season {year: 2021})
MATCH (res)-[:BY_DRIVER]->(d:Driver)
WITH d,
     sum(coalesce(res.points, 0.0)) AS Points,
     sum(CASE WHEN res.position = 1 THEN 1 ELSE 0 END) AS Wins
RETURN d.forename + ' ' + d.surname AS Driver, Points, Wins
ORDER BY Points DESC, Wins DESC, Driver ASC

//2. Query 2
MATCH (res:Result)-[:FOR_RACE]->(r:Race)
MATCH (res)-[:FOR_CONSTRUCTOR]->(c:Constructor)
MATCH (res)-[:HAS_STATUS]->(s:Status)
WHERE r.year >= 2014 AND r.year <= 2021
WITH c,
     count(res) AS Entries,
     sum(CASE WHEN s.status = 'Finished' OR s.status STARTS WITH '+' THEN 1 ELSE 0 END) AS Classified
WHERE Entries >= 100
RETURN c.constructor_name AS Constructor, Entries, Classified,
       toFloat(Classified) / Entries AS Reliability
ORDER BY Reliability DESC, Constructor ASC

//3. Query 3
MATCH (h:Result)-[:BY_DRIVER]->(:Driver {driverId: 1})
MATCH (b:Result)-[:BY_DRIVER]->(:Driver {driverId: 822})
MATCH (h)-[:FOR_RACE]->(r:Race)<-[:FOR_RACE]-(b)
MATCH (h)-[:FOR_CONSTRUCTOR]->(c:Constructor)<-[:FOR_CONSTRUCTOR]-(b)
WITH c, r, min(h.positionOrder) AS hPosition, min(b.positionOrder) AS bPosition
RETURN c.constructor_name AS Constructor,
       count(*) AS Shared_Races,
       sum(CASE WHEN hPosition < bPosition THEN 1 ELSE 0 END) AS Hamilton_Ahead,
       sum(CASE WHEN bPosition < hPosition THEN 1 ELSE 0 END) AS Bottas_Ahead
ORDER BY Constructor ASC

//4. Query 4
MATCH (res:Result)-[:FOR_RACE]->(r:Race)-[:AT_CIRCUIT]->(:Circuit {circuitId: 6})
MATCH (res)-[:BY_DRIVER]->(d:Driver)
MATCH (res)-[:HAS_STATUS]->(s:Status)
WHERE r.date < date('2020-01-01')
  AND NOT s.status IN ['Did not qualify', 'Did not prequalify', 'Withdrew']
WITH d,
     count(DISTINCT r) AS Starts,
     avg(res.positionOrder) AS avgPos,
     sum(CASE WHEN res.position = 1 THEN 1 ELSE 0 END) AS Wins
WHERE Starts >= 5
WITH d, Starts, round(avgPos, 4) AS Avg_Finish, Wins
RETURN d.forename + ' ' + d.surname AS Driver, Starts, Avg_Finish, Wins
ORDER BY Avg_Finish ASC, Wins DESC, Driver ASC

//5. Query 5
MATCH (res:Result)-[:FOR_RACE]->(r:Race)
WHERE r.year IN [2020, 2021]
MATCH (res)-[:BY_DRIVER]->(d:Driver)
MATCH (res)-[:FOR_CONSTRUCTOR]->(c:Constructor)
WITH d, r.year AS Season, c, count(res) AS Entries
ORDER BY Entries DESC, c.constructorId ASC
WITH d, Season, collect(c)[0] AS PrimaryConstructor
WITH d,
     collect(CASE WHEN Season = 2020 THEN PrimaryConstructor END)[0] AS c2020,
     collect(CASE WHEN Season = 2021 THEN PrimaryConstructor END)[0] AS c2021
WHERE c2020 IS NOT NULL AND c2021 IS NOT NULL
  AND c2020.constructorId <> c2021.constructorId
WITH d.forename + ' ' + d.surname AS Driver
ORDER BY Driver ASC
RETURN count(Driver) AS Drivers_Changed, collect(Driver) AS Drivers
